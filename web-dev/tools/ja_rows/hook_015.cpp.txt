// Batch harness: once the story's text object (28 cells, 1392 wide) has had a text set, every line of NX_LAYOUT_BATCH ("id<TAB>text[<TAB>startx]") is set on
// it in turn and the laid-out buffer written to NX_LAYOUT_OUT.
std::string g_layout_id;

    if (*(int*)(obj + 0x108b4) != wantCap) return;

        if (body.size() >= 0xff0) continue;

        memcpy(buf, body.data(), body.size());

        c.x[0] = obj; c.x[1] = (uint64_t)buf; c.x[2] = flags; c.v[0] = v0;
